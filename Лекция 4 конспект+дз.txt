# pandas конкатенация
ser1 = pd.Series(['a', 'b', 'c'], index = [1, 2, 3])
ser2 = pd.Series(['d', 'e', 'f'], index = [4, 5, 6])

print (pd.concat([ser1, ser2]))

# индексы могут дублироваться но это помешает доступу к ним
ser1 = pd.Series(['a', 'b', 'c'], index = [1, 2, 3])
ser2 = pd.Series(['d', 'e', 'f'], index = [1, 2, 6])

print (pd.concat([ser1, ser2])) # OK
# если такое не нужно то
print (pd.concat([ser1, ser2], verify_integrity=False)) # True = NOT OK

print (pd.concat([ser1, ser2], ignore_index=True)) # True = Перезапись индексов

print (pd.concat([ser1, ser2], keys = ['x', 'y'])) # keys - создание индексов

# Задание №4 Пример inner и outer джоинов для Series
ser1 = pd.Series(['a', 'b', 'c'], index = [1, 2, 3])
ser2 = pd.Series(['b', 'c', 'f'], index = [1, 5, 3])
# все индексы объединяются, а недостающие добавляются со значением NaN
print (pd.concat([ser1, ser2], axis=1, join = 'outer'))
# объединяются все совпадающие в серии индексы, а недостающие не добавляются
print (pd.concat([ser1, ser2], axis=1, join = 'inner'))
